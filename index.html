🏔 Cloudflare R2 — Large Files Recommended for 3D artists

Built for huge files — multi-GB ZBrush sculpts, Blender files with rigs and packed textures, anything. Stratum automatically splits big files into chunks and reassembles them, so even a 4GB .ztl uploads reliably. 10GB free, downloads always free, then just $0.015/GB per month (100GB ≈ $1.50/mo).

▶ Setup guide (~5 min, free)
1. dash.cloudflare.com → R2 → Create bucket (name it e.g. stratum-files)
2. Workers & Pages → Create Worker → paste the code below → Deploy
3. Worker → Settings → Bindings → Add → R2 bucket → Variable name BUCKET → pick your bucket
4. (Recommended) Settings → Variables → add AUTH_KEY = any password you invent. Enter the same below so only you can upload.
5. Paste the worker URL below. Done — drag in your gigabytes. 
export default {
  async fetch(request, env) {
    const cors = {
      'Access-Control-Allow-Origin': '*',
      'Access-Control-Allow-Methods': 'GET, PUT, POST, DELETE, OPTIONS',
      'Access-Control-Allow-Headers': '*',
    };
    if (request.method === 'OPTIONS')
      return new Response(null, { headers: cors });

    const url = new URL(request.url);
    const key = url.searchParams.get('key');
    const action = url.searchParams.get('action');
    const json = (o, s = 200) => new Response(JSON.stringify(o),
      { status: s, headers: { ...cors, 'Content-Type': 'application/json' } });

    // Auth for anything that writes (GET stays public for downloads)
    if (request.method !== 'GET' && env.AUTH_KEY) {
      if (request.headers.get('X-Auth-Key') !== env.AUTH_KEY)
        return json({ error: 'Unauthorized' }, 401);
    }
    if (!key) return json({ error: 'Missing key' }, 400);

    // Download (streams from R2 — free egress)
    if (request.method === 'GET') {
      const obj = await env.BUCKET.get(key);
      if (!obj) return json({ error: 'Not found' }, 404);
      return new Response(obj.body, { headers: { ...cors,
        'Content-Type': obj.httpMetadata?.contentType || 'application/octet-stream',
        'Content-Length': obj.size,
        'Content-Disposition': `attachment; filename="${key.split('/').pop()}"` } });
    }
    // Delete a version
    if (request.method === 'DELETE') {
      await env.BUCKET.delete(key);
      return json({ ok: true });
    }
    // Small file: single PUT
    if (request.method === 'PUT' && !action) {
      await env.BUCKET.put(key, request.body, { httpMetadata:
        { contentType: request.headers.get('Content-Type') || 'application/octet-stream' } });
      return json({ ok: true, key });
    }
    // Multipart for big files
    if (request.method === 'POST' && action === 'create') {
      const up = await env.BUCKET.createMultipartUpload(key, { httpMetadata:
        { contentType: request.headers.get('Content-Type') || 'application/octet-stream' } });
      return json({ uploadId: up.uploadId });
    }
    if (request.method === 'PUT' && action === 'part') {
      const up = env.BUCKET.resumeMultipartUpload(key, url.searchParams.get('uploadId'));
      const part = await up.uploadPart(parseInt(url.searchParams.get('part')), request.body);
      return json({ partNumber: part.partNumber, etag: part.etag });
    }
    if (request.method === 'POST' && action === 'complete') {
      const up = env.BUCKET.resumeMultipartUpload(key, url.searchParams.get('uploadId'));
      await up.complete(await request.json());
      return json({ ok: true, key });
    }
    if (request.method === 'POST' && action === 'abort') {
      const up = env.BUCKET.resumeMultipartUpload(key, url.searchParams.get('uploadId'));
      await up.abort();
      return json({ ok: true });
    }
    return json({ error: 'Bad request' }, 400);
  }
};

☁ Option 2 — Cloudinary Direct All file types

Cloudinary supports any file type including .blend, .fbx, .psd, .obj. Free up to 25GB. Direct uploads from your browser work as long as your upload preset is set to Unsigned.

Setup:
1. cloudinary.com → Sign up free
2. Dashboard → Settings → Upload tab → Add upload preset
3. Set Signing mode: Unsigned — this is the key step!
4. Copy the preset name and your Cloud Name (shown at top of dashboard)
🔧 Option 3 — Cloudinary via Worker Proxy Advanced

If direct Cloudinary uploads don't work (some networks block them), a Cloudflare Worker acts as a proxy. Deploy it once for free in ~2 minutes.

▶ Show Worker code (copy & deploy to workers.cloudflare.com) 
export default {
  async fetch(request) {
    if (request.method === 'OPTIONS')
      return new Response(null, { headers: corsHeaders() });
    if (request.method !== 'POST')
      return new Response('Method not allowed', { status: 405 });
    const form = await request.formData();
    const cloudName = form.get('cloud_name');
    const body = new FormData();
    body.append('file', form.get('file'));
    body.append('upload_preset', form.get('upload_preset'));
    body.append('folder', form.get('folder') || 'stratum');
    const res = await fetch(
      `https://api.cloudinary.com/v1_1/${cloudName}/auto/upload`,
      { method: 'POST', body }
    );
    const data = await res.json();
    return new Response(JSON.stringify(data), {
      headers: { 'Content-Type': 'application/json', ...corsHeaders() }
    });
  }
};
function corsHeaders() {
  return {
    'Access-Control-Allow-Origin': '*',
    'Access-Control-Allow-Methods': 'POST, OPTIONS',
    'Access-Control-Allow-Headers': '*',
  };
}
